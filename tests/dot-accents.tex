\section*{Gaussian Modern: dot accents and primes}
Three and four dots extend the existing double-dot accent.
\[
\begin{array}{ccccc}
\dotrow{x} \\[8pt]
\dotrow{f} \\[8pt]
\dotrow{X} \\[8pt]
\dotrow{\alpha} \\[8pt]
\dotrow{\imath} \\[8pt]
\dotrow{\symbf{x}} \\[8pt]
\dotrow{\symbfit{x}}
\end{array}
\]
\subsection*{Superscripts and nested superscripts}
\[
a^{\dot{x}}\quad a^{\ddot{x}}\quad a^{\dddot{x}}\quad a^{\ddddot{x}}
\qquad a^{b^{\dddot{x}}}\quad a^{b^{\ddddot{x}}}
\]
\subsection*{Subscripts, hats and wider expressions}
\[
a_{\dddot{x}}\quad a_{\ddddot{x}}\qquad
\hat{x}\quad\dddot{xy}\quad\ddddot{xy}\quad\widehat{xyz}
\]
\subsection*{At ordinary text size}
The derivatives $\dot{x}$, $\ddot{x}$, $\dddot{x}$ and $\ddddot{x}$
share the height and spacing of the existing accents.
